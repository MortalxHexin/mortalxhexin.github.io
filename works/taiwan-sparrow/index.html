會抖的墨線

每一筆都沿著線條的法線推一點雜訊，每秒換 12 次，線條就像手繪一樣微微顫動。p 從 0 走到 1，就是一筆一筆畫出來。

// 沿著線條的法線方向推一點點雜訊，像手在抖
const r = pts.map((p, i) => {
  const [nx, ny] = normal(pts, i);
  const o = noise(i * .09, seed + boil) * wob
          + noise(i * .5,  seed + boil + 3) * wob * .35;
  return [p[0] + nx * o, p[1] + ny * o];
});
// boil = Math.floor(t * 12)：每秒換 12 次種子，線條會「呼吸」
// p：畫到哪裡為止，0 → 1 就是一筆一筆畫出來
drawPolyline(r.slice(0, r.length * p));

把蚊香點著

沿著螺旋算出麻雀滑過的路徑，走過的地方畫成發光的引信，再從引信上冒出煙。畫面同時跟著螺旋轉一點角度，製造暈眩感。

function spiral(k) {                    // k：0 在中心，1 在尾端
  const th = lerp(.96 - 5 * Math.PI, .96, k);
  const r  = lerp(40, 420, k);
  return [cx + r * Math.cos(th), cy + r * .62 * Math.sin(th)];
}
const burnt = range(0, 160 * k).map(i => spiral(i / 160));
stroke(burnt, '#ff9a3c', 38, .4);       // 外層光暈
stroke(burnt, '#ff6a1f', 10, .95);      // 引信本體
